import {beforeAll,afterAll,it,expect,vi} from 'vitest';
import pg from 'pg';
import {randomUUID} from 'node:crypto';
import {Database} from '../src/db/database';
import {allocateBudget,authorizeTool,settleTool,increaseRunBudget,currentPeriod} from '../src/budget';
import {micros,decimal} from '../src/budget/money';
const admin=new pg.Pool({connectionString:'postgres://postgres@127.0.0.1:55432/reinlab_test'});
const pool=new pg.Pool({connectionString:'postgres://reinlab_app@127.0.0.1:55432/reinlab_test',max:12}),db=new Database(pool);
const A=randomUUID(),B=randomUUID();
beforeAll(async()=>{await db.assertSafeRole();await admin.query('insert into account(id,monthly_budget_cny) values($1,9999),($2,9999)',[A,B]);});
afterAll(async()=>{await pool.end();await admin.end();});
async function fixture(amount='10') {const d=randomUUID(),id=randomUUID();await admin.query("insert into dossier(id,account_id,bench,title,workspace_path) values($1,$2,'research','独立预算核验',$3)",[d,A,'/data/dossiers/'+d]);await admin.query("insert into run(id,account_id,dossier_id,role_id,mode,request_id,state,budget_cny) values($1,$2,$3,'research-read','read',$4,'running',$5)",[id,A,d,randomUUID(),amount]);const root=await db.withAccount(A,c=>allocateBudget(c,A,{dossierId:d,amountCny:amount}));const allocation=await db.withAccount(A,c=>allocateBudget(c,A,{dossierId:d,amountCny:amount,parentId:root.id,runId:id}));return {id,d,root,allocation};}
async function child(r:Awaited<ReturnType<typeof fixture>>,budget='8') {const id=randomUUID();await admin.query("insert into run(id,account_id,dossier_id,parent_run_id,role_id,mode,request_id,state,budget_cny) values($1,$2,$3,$4,'swarm-reader','swarm',$5,'running',$6)",[id,A,r.d,r.id,randomUUID(),budget]);return id;}
const hold=(id:string,amount:string,key=randomUUID())=>db.withAccount(A,c=>authorizeTool(c,A,{runId:id,requestId:key,maximumCny:amount}));
const pay=(id:string,amount:string|null)=>db.withAccount(A,c=>settleTool(c,A,{holdId:id,costCny:amount,credentialKind:'api_key'}));
it('独立六位定点边界及不精确JS数值拒绝',()=>{for(const x of ['0','0.000001','999999.999999','001.200000','1e-6'])expect(decimal(micros(x))).toMatch(/^\d+\.\d{6}$/);for(const x of ['1000000','1e309','1e-999','-0.1',0.1+0.2])expect(()=>micros(x)).toThrow();});
it('独立并发十工具仅一份尾额获准且拒绝后不生成hold',async()=>{const r=await fixture('1'),h=await hold(r.id,'0.9');await pay(h.holdId!,'0.9');const attempts=await Promise.all(Array.from({length:10},()=>hold(r.id,'0.1')));expect(attempts.filter(x=>x.allowed)).toHaveLength(1);expect((await admin.query('select count(*)::int n from budget_hold where run_id=$1',[r.id])).rows[0].n).toBe(2);});
it('独立并发同request只生成一保留和一费用事件',async()=>{const r=await fixture(),key=randomUUID(),holds=await Promise.all(Array.from({length:6},()=>hold(r.id,'1',key)));expect(new Set(holds.map(h=>h.holdId)).size).toBe(1);await Promise.all(Array.from({length:6},()=>pay(holds[0].holdId!,'0.123456')));expect((await admin.query('select spent_cny from run where id=$1',[r.id])).rows[0].spent_cny).toBe('0.123456');expect((await admin.query("select count(*)::int n from event where run_id=$1 and kind='cost'",[r.id])).rows[0].n).toBe(1);});
it('独立新增三表跨账户写入与读取隔离',async()=>{const r=await fixture(),h=await hold(r.id,'1');for(const table of ['budget_estimate','budget_allocation','budget_hold']){expect((await db.withAccount(B,c=>c.query(`select * from ${table} where account_id=$1`,[A]))).rowCount).toBe(0);expect((await db.withAccount(B,c=>c.query(`update ${table} set account_id=$1 where account_id=$2`,[B,A])).catch(()=>({rowCount:0}))).rowCount).toBe(0);}await expect(db.withAccount(B,c=>c.query("insert into budget_estimate(account_id,dossier_id,mode,model_tier,quote,expires_at) values($1,$2,'read','strong','{}',now()+interval '1 minute')",[A,r.d]))).rejects.toThrow();await expect(db.withAccount(B,c=>c.query('insert into budget_allocation(account_id,dossier_id,period,amount_cny) values($1,$2,$3,1)',[A,r.d,currentPeriod()]))).rejects.toThrow();await expect(db.withAccount(B,c=>c.query("insert into budget_hold(account_id,dossier_id,run_id,allocation_id,period,request_id,maximum_cny,maximum_quota,maximum_gpu_seconds,state) values($1,$2,$3,$4,$5,$6,1,0,0,'reserved')",[A,r.d,r.id,r.allocation.id,currentPeriod(),randomUUID()]))).rejects.toThrow();expect(h.allowed).toBe(true);});
it('独立hold来源不能引用同账户其他运行allocation',async()=>{const r=await fixture(),s=await fixture();await expect(db.withAccount(A,c=>c.query("insert into budget_hold(account_id,dossier_id,run_id,allocation_id,period,request_id,maximum_cny,maximum_quota,maximum_gpu_seconds,state) values($1,$2,$3,$4,$5,$6,1,0,0,'reserved')",[A,r.d,r.id,s.allocation.id,currentPeriod(),randomUUID()]))).rejects.toThrow();});
it('独立旧月reserved不结清不能新月确认',async()=>{const n=new Date(),clock=vi.spyOn(Date,'now').mockReturnValue(Date.UTC(n.getUTCFullYear(),n.getUTCMonth()-1,15));let r:Awaited<ReturnType<typeof fixture>>;try{r=await fixture();await hold(r.id,'1');}finally{clock.mockRestore();}await expect(db.withAccount(A,c=>increaseRunBudget(c,A,r!.id,'5'))).rejects.toMatchObject({code:'USAGE_PENDING'});expect((await admin.query('select count(*)::int n from budget_allocation where run_id=$1',[r!.id])).rows[0].n).toBe(1);});
it('独立旧月unpriced不结清不能新月确认',async()=>{const n=new Date(),clock=vi.spyOn(Date,'now').mockReturnValue(Date.UTC(n.getUTCFullYear(),n.getUTCMonth()-1,15));let r:Awaited<ReturnType<typeof fixture>>;try{r=await fixture();const h=await hold(r.id,'1');await pay(h.holdId!,null);}finally{clock.mockRestore();}await expect(db.withAccount(A,c=>increaseRunBudget(c,A,r!.id,'5'))).rejects.toMatchObject({code:'USAGE_PENDING'});});
it('独立子预算不能占用父工具已保留的费用',async()=>{const r=await fixture(),h=await hold(r.id,'8'),id=await child(r,'8');expect(h.allowed).toBe(true);await expect(db.withAccount(A,c=>allocateBudget(c,A,{dossierId:r.d,amountCny:'8',parentId:r.allocation.id,runId:id}))).rejects.toMatchObject({code:'BUDGET_EXCEEDED'});});
it('独立追加子预算不能占用父工具已保留费用，应增长祖先',async()=>{const r=await fixture(),id=await child(r,'2');await db.withAccount(A,c=>allocateBudget(c,A,{dossierId:r.d,amountCny:'2',parentId:r.allocation.id,runId:id}));const h=await hold(r.id,'8');expect(h.allowed).toBe(true);await db.withAccount(A,c=>increaseRunBudget(c,A,id,'3'));expect((await admin.query('select amount_cny from budget_allocation where id=$1',[r.root.id])).rows[0].amount_cny).toBe('13.000000');expect((await admin.query('select budget_cny from run where id=$1',[r.id])).rows[0].budget_cny).toBe('13.000000');});
it('独立祖先旧月有子运行在途费用不能建立新月划拨',async()=>{const n=new Date(),clock=vi.spyOn(Date,'now').mockReturnValue(Date.UTC(n.getUTCFullYear(),n.getUTCMonth()-1,15));let r:Awaited<ReturnType<typeof fixture>>;try{r=await fixture();const id=await child(r,'8');await db.withAccount(A,c=>allocateBudget(c,A,{dossierId:r.d,amountCny:'8',parentId:r.allocation.id,runId:id}));await hold(id,'1');}finally{clock.mockRestore();}await expect(db.withAccount(A,c=>increaseRunBudget(c,A,r!.id,'5'))).rejects.toMatchObject({code:'USAGE_PENDING'});});
it('独立三张新增表撤销DELETE/TRUNCATE且估算不可修改',async()=>{const rows=await admin.query("select table_name,privilege_type from information_schema.role_table_grants where grantee='reinlab_app' and table_name in ('budget_estimate','budget_allocation','budget_hold')");for(const t of ['budget_estimate','budget_allocation','budget_hold']){const permissions=rows.rows.filter(r=>r.table_name===t).map(r=>r.privilege_type);expect(permissions).not.toContain('DELETE');expect(permissions).not.toContain('TRUNCATE');if(t==='budget_estimate')expect(permissions).not.toContain('UPDATE');}});
it('独立UTC月份只按UTC边界切换',()=>{const clock=vi.spyOn(Date,'now');try{clock.mockReturnValue(Date.UTC(2026,9,31,23,59,59));expect(currentPeriod()).toBe('2026-10-01');clock.mockReturnValue(Date.UTC(2026,10,1,0,0,0));expect(currentPeriod()).toBe('2026-11-01');}finally{clock.mockRestore();}});
it('独立父运行未知费用保留仍不可划给子运行',async()=>{const r=await fixture(),h=await hold(r.id,'8');await pay(h.holdId!,null);const id=await child(r,'8');await expect(db.withAccount(A,c=>allocateBudget(c,A,{dossierId:r.d,amountCny:'8',parentId:r.allocation.id,runId:id}))).rejects.toMatchObject({code:'BUDGET_EXCEEDED'});});
it('独立祖先旧月有子运行未知费用不能建立新月划拨',async()=>{const n=new Date(),clock=vi.spyOn(Date,'now').mockReturnValue(Date.UTC(n.getUTCFullYear(),n.getUTCMonth()-1,15));let r:Awaited<ReturnType<typeof fixture>>;try{r=await fixture();const id=await child(r,'8');await db.withAccount(A,c=>allocateBudget(c,A,{dossierId:r.d,amountCny:'8',parentId:r.allocation.id,runId:id}));const h=await hold(id,'1');await pay(h.holdId!,null);}finally{clock.mockRestore();}await expect(db.withAccount(A,c=>increaseRunBudget(c,A,r!.id,'5'))).rejects.toMatchObject({code:'USAGE_PENDING'});});
